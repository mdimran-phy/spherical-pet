cmake_minimum_required(VERSION 3.16)
project(spherical-pet VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(WITH_GEANT4_UIVIS "Enable Geant4 interactive UI and visualization" OFF)
if(WITH_GEANT4_UIVIS)
  find_package(Geant4 11.1 REQUIRED ui_all vis_all)
else()
  find_package(Geant4 11.1 REQUIRED)
endif()
include(${Geant4_USE_FILE})
find_package(Boost REQUIRED COMPONENTS date_time program_options filesystem system regex)
set(JANSSON_BUILD_DOCS OFF CACHE BOOL "" FORCE)
set(JANSSON_BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(JANSSON_WITHOUT_TESTS ON CACHE BOOL "" FORCE)
include(CTest)
# Compatibility with the pinned Jansson subproject on newer CMake releases.
set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
add_subdirectory(third_party/jansson EXCLUDE_FROM_ALL)
file(GLOB_RECURSE SIPM_SOURCES CONFIGURE_DEPENDS third_party/g4sipm/src/*.cc)
add_library(pet_g4sipm STATIC ${SIPM_SOURCES})
target_include_directories(pet_g4sipm PUBLIC third_party/g4sipm/include
  third_party/jansson/src ${CMAKE_CURRENT_BINARY_DIR}/third_party/jansson/include)
target_link_libraries(pet_g4sipm PUBLIC ${Geant4_LIBRARIES} ${Boost_LIBRARIES} jansson)
add_executable(spherical-pet src/main.cc src/Config.cc src/DetectorConstruction.cc)
target_include_directories(spherical-pet PRIVATE include)
target_link_libraries(spherical-pet PRIVATE pet_g4sipm)
if(WITH_GEANT4_UIVIS)
  target_compile_definitions(spherical-pet PRIVATE PET_WITH_UIVIS=1)
endif()
include(GNUInstallDirs)
install(TARGETS spherical-pet RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(DIRECTORY config macros scripts docs DESTINATION ${CMAKE_INSTALL_DATADIR}/spherical-pet
        PATTERN "__pycache__" EXCLUDE)
install(FILES README.md LICENSE THIRD_PARTY_NOTICES.md requirements.txt
        DESTINATION ${CMAKE_INSTALL_DATADIR}/spherical-pet)
include(CTest)
if(BUILD_TESTING)
  add_executable(test-config src/Config.cc tests/test_config.cc)
  target_include_directories(test-config PRIVATE include)
  add_test(NAME config_unit COMMAND test-config)
  find_package(Python3 COMPONENTS Interpreter)
  if(Python3_Interpreter_FOUND)
    add_test(NAME cli COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_cli.py $<TARGET_FILE:spherical-pet>)
    add_test(NAME transport COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_transport.py $<TARGET_FILE:spherical-pet>)
    set_tests_properties(transport PROPERTIES TIMEOUT 180)
  endif()
endif()
